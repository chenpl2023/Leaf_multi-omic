setwd("E:/陈沛霖-病原菌/代谢-微生物组/code/Leaf_multi-omic1")
library(ggplot2)
library(tidyverse)
library(reshape2)
betadis_otu <- read.csv("Leaf_fungi_betadisper_box.csv",header = T)
betadis_trans <- read.csv("Leaf_trans_betadisper_box.csv",header = T)
betadis_metab <- read.csv("Leaf_metab_betadisper_box.csv",header = T)
colnames(betadis_otu)[1] <- 'OTU_disper'
colnames(betadis_trans)[1] <- 'Transcript_disper'
colnames(betadis_metab)[1] <- 'Metab_disper'
betadis_otu$lev<-interaction(betadis_otu$Treat, betadis_otu$Week,sep = ":") 
betadis_trans$lev<-interaction(betadis_trans$Treat, betadis_trans$Week,sep = ":")
betadis_metab$lev<-interaction(betadis_metab$Treat, betadis_metab$Week,sep = ":")

betadis_metab <- left_join(betadis_metab,betadis_trans[,c(1,5)],by='lev')
betadis_metab <- left_join(betadis_metab,betadis_otu[,c(1,5)],by='lev')


a <- ggplot(data = betadis_metab, aes(x = Transcript_disper,  y = Metab_disper)) +
  geom_point(aes(colour=Week,shape=Treat), size = 2, alpha = 1)+
  geom_smooth(se=FALSE,size=1, method="lm", aes(), alpha = 0.2,color = 'grey80')+
  ggpubr::stat_cor(aes(label = paste(..r.label..,..p.label.., sep = "~`,`~")),label.x = 0.2,size=4,fontface='bold',col='black')+
  labs(x="Metatranscriptomic beta diversity",y="Metabolomic beta diversity")+
  theme_bw()+
  scale_color_distiller(palette = "Spectral")+ scale_shape_manual(values=c(15,17,19))+
  theme(plot.title=element_text(face="bold",hjust=0.5,size=10),
        strip.text = element_text(size = 10,face="bold"),
        panel.spacing = unit(0, "lines"),
        legend.title = element_text(colour="black", size=10, face="bold"),
        legend.text = element_text(colour="black", size=9, face="bold"),
        axis.text=element_text(size=8,face="bold"),
        axis.title=element_text(size=10,face="bold"),title=element_text(size=10,face="bold"),legend.position = "bottom")

b <- ggplot(data = betadis_metab, aes(x = OTU_disper,  y = Metab_disper)) +

  geom_smooth(se=FALSE,size=1, method="lm", aes(), alpha = 0.2,color = 'grey80')+

  labs(x="Taxonomic beta diversity",y="Metabolomic beta diversity")+
  theme_bw()+
  scale_color_distiller(palette = "Spectral")+ scale_shape_manual(values=c(15,17,19))+
  theme(plot.title=element_text(face="bold",hjust=0.5,size=10),
        strip.text = element_text(size = 10,face="bold"),
        panel.spacing = unit(0, "lines"),
        legend.title = element_text(colour="black", size=10, face="bold"),
        legend.text = element_text(colour="black", size=9, face="bold"),
        axis.text=element_text(size=8,face="bold"),
        axis.title=element_text(size=10,face="bold"),title=element_text(size=10,face="bold"),legend.position = "bottom")


c <- ggplot(data = betadis_metab, aes(y = Metab_disper,x =Week)) +
  geom_point(aes(colour=Week,shape=Treat), size = 2, alpha = 1)+
  scale_shape_manual(values=c(15,17,19))+
  geom_smooth(se=FALSE,size=1, method="lm", aes(), alpha = 0.2,col='grey80')+
  ggpubr::stat_cor(aes(label = paste(..r.label..,..p.label.., sep = "~`,`~")),label.x = 3,size=4,fontface='bold')+

library(ggpubr)
ggarrange(c,b,a,ncol=3,common.legend = TRUE,legend = "bottom")
