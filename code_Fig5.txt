zg <- read.csv("Group.csv",header = T,row.names = 1)
nrow(zg)
zg <- subset(zg, ID_metab != '#N/A')
nrow(zg)
zg <- subset(zg, Habitat == 'Leaf')
otu <- read.csv("0000BacteriaFungi.csv",header = T,row.names = 1)
fungi <- subset(otu, otu$Kingdom1=="Fungi")
fungi <- fungi[,(1:1029)]
fungi <- data.frame(t(fungi))
fungi <- as.data.frame(fungi[row.names(fungi)%in%zg$ID_metab,])

fungi = as.data.frame(rrarefy(fungi, min(rowSums(fungi))))

Trans <- as.data.frame(Trans[row.names(Trans)%in%row.names(zg),])
Trans$ID = row.names(Trans)
zg$ID = row.names(zg)
library(dplyr)
Trans_g <- left_join(Trans,zg[,c(5,11)],by='ID')
row.names(Trans_g) <- Trans_g$ID_metab
Trans_g <- select(Trans_g,-c(ID,ID_metab))
ncol(Trans_g)
Trans_g <-Trans_g[order(row.names(Trans_g)),]
fungi <-fungi[order(row.names(fungi)),]
row.names(Trans_g)==row.names(fungi)

Sorg_leaf_trans <- read.table("leaf_normCounts.txt",header=T, sep='\t',row.names = 1)
Sorg_leaf_trans <- data.frame(t(Sorg_leaf_trans))
Sorg_leaf_trans$ID3 = row.names(Sorg_leaf_trans)
Sorg_leaf_trans <- left_join(zg[,c(5,3)],Sorg_leaf_trans,by='ID3')
row.names(Sorg_leaf_trans) <- Sorg_leaf_trans$ID_metab
Sorg_leaf_trans <- select(Sorg_leaf_trans,-c(ID3,ID_metab))
Sorg_leaf_trans <-Sorg_leaf_trans[order(row.names(Sorg_leaf_trans)),]
row.names(Trans_g)==row.names(Sorg_leaf_trans)

net_L <- cbind(Trans_g,fungi,Sorg_leaf_trans,metab)

library(ade4)

w2 <- ktab.data.frame(w1, df$number, tabnames = as.character(df$type))

row.names(coolig) == row.names(aa)
